# Reproducible desktop runtime profile. Build on the target OS.
#
# These pins mirror the versions the application is developed and run against,
# so a frozen build behaves like a source checkout. Every pin below publishes
# cp313 wheels for win_amd64 and manylinux x86_64 — the desktop build must never
# need a compiler on the release runners.
Pillow==12.3.0
pillow-heif==1.5.0
ImageHash==4.3.2
pyexiftool==0.5.6
numpy==2.5.1
# The headless variant: same OpenCV version and sources as opencv-python, built
# without the highgui backend. Nothing in the shipped package uses a GUI or video
# symbol — cv2 appears only as cvtColor/resize/Laplacian/dnn, and video frames come
# from the bundled ffmpeg, not VideoCapture — so the Qt5 Core/Gui/Widgets/Test/XcbQpa
# and xcb/xkb stack the full wheel carries is 29 MB of bundle nobody can reach.
# If a cv2.imshow/VideoCapture ever lands in organize_archive/, this pin has to go
# back to opencv-python; tests/unit/test_opencv_headless.py fails the build if it does.
opencv-python-headless==5.0.0.93
onnxruntime==1.27.0
scikit-learn==1.9.0
# k-NN for the face cluster core/expansion passes. faces/cluster.py degrades to a
# blocked-GEMM path when this is missing, so a build without it stays correct and
# only gets slower — which is exactly why its absence went unnoticed until now.
# The wheel carries a bundled libfaiss; organize-archive.spec collects it.
faiss-cpu==1.14.3
# Face detection/recognition backend (SCRFD det_10g + ArcFace w600k_r50 from the
# buffalo_l pack). Pure-Python wheel (py3-none-any): it only supplies the model
# zoo loader and face_align helpers, inference itself runs on onnxruntime.
insightface==1.0.1
# Query tokenizer for local semantic search (SigLIP 2's Gemma SentencePiece
# vocabulary). A Rust extension, but its wheels are cp310-abi3 for both
# manylinux x86_64 and win_amd64, so cp313 installs one without a compiler.
tokenizers==0.23.1
PyInstaller==6.21.0
