# Reproducible desktop runtime profile. Build on the target OS.
#
# These pins mirror the versions the application is developed and run against,
# so a frozen build behaves like a source checkout. Every pin below publishes
# cp313 wheels for win_amd64 and manylinux x86_64 — the desktop build must never
# need a compiler on the release runners.
Pillow==12.3.0
pillow-heif==1.5.0
ImageHash==4.3.2
pyexiftool==0.5.6
numpy==2.5.1
# The headless variant: same OpenCV version and sources as opencv-python, built
# without the highgui backend. Nothing in the shipped package uses a GUI or video
# symbol — cv2 appears only as cvtColor/resize/Laplacian/dnn, and video frames come
# from the bundled ffmpeg, not VideoCapture — so the Qt5 Core/Gui/Widgets/Test/XcbQpa
# and xcb/xkb stack the full wheel carries is 30 MB of bundle nobody can reach.
# If a cv2.imshow/VideoCapture ever lands in trove/, this pin has to go
# back to opencv-python; tests/unit/test_opencv_headless.py fails the build if it does.
#
# Pinning it is not enough to get it. Two packages listed below -- insightface
# and rapidocr -- require plain opencv-python, so pip satisfies every line in
# this file by installing *both* wheels; they write the same cv2/, and the last
# one unpacked is the one that gets frozen. 0.3.0 shipped the wrong one. Run
# packaging/scripts/ensure-headless-opencv.py after installing this file.
opencv-python-headless==5.0.0.93
onnxruntime==1.27.0
scikit-learn==1.9.0
# Face detection/recognition backend (SCRFD det_10g + ArcFace w600k_r50 from the
# buffalo_l pack). Pure-Python wheel (py3-none-any): it only supplies the model
# zoo loader and face_align helpers, inference itself runs on onnxruntime.
insightface==1.0.1
# Query tokenizer for local semantic search (SigLIP 2's Gemma SentencePiece
# vocabulary). A Rust extension, but its wheels are cp310-abi3 for both
# manylinux x86_64 and win_amd64, so cp313 installs one without a compiler.
tokenizers==0.23.1
# PDF text extraction and page rasterising, for the Documents feature. Ships a
# prebuilt PDFium; its wheels are py3-none-<platform>, so cp313 needs no build.
pypdfium2==5.12.1
# OCR: PP-OCRv6 detection and recognition. The wheel carries the ONNX weights
# too, but packaging/trove.spec filters them out and the app fetches them on
# first use instead -- they were 26.5 MB of every installer, the largest single
# item in it, for a feature most users never switch on (ADR 0019). What is left
# of this package after the weights is 0.1 MB.
rapidocr==3.9.2
pyclipper==1.4.0
shapely==2.1.2
# Filesystem notifications, so files added to an archive are noticed in seconds
# rather than on the next sweep. A Rust extension over the `notify` crate, with
# cp313 wheels for both win_amd64 and manylinux x86_64, so it needs no compiler
# on the release runners. The app runs correctly without it -- the sweep is what
# is authoritative -- so a build that somehow lacked it would be slower to notice
# a new folder of photos and nothing else.
watchfiles==1.2.0
PyInstaller==6.21.0
